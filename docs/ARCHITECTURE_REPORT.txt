## SECTION 5: AUTH & MIDDLEWARE

### Middleware (`middleware.ts`)
- **Location:** `/middleware.ts` (project root)
- **Functionality:**
  - Enforces canonical host (ozrentaplane.com → www.ozrentaplane.com redirect)
  - Creates Supabase server client for session refresh
  - Calls `getUser()` to refresh JWT tokens on every request
  - Runs on all routes except Next.js internals and static assets
- **No explicit role checks** — role-based access is handled in page/component level via:
  - `redirect('/admin')` for admins accessing `/dashboard`
  - `redirect('/dashboard')` for customers accessing `/admin`

### Authentication Flow
- **Provider:** Supabase Auth (OAuth + Email)
- **OAuth callback:** `/app/auth/callback/route.ts`
- **Session storage:** Secure HTTP-only cookies (managed by `@supabase/ssr`)
- **Profile auto-creation:** Trigger function `handle_new_user()` creates profile on signup with role = 'customer'

### Role-Based Access Control
- **Two roles:** 'customer' | 'admin'
- **Access enforcement points:**
  - **Middleware:** Page redirects based on role checks
  - **Server actions:** `requireCustomer()` and `requireAdmin()` guard functions verify role before operations
  - **RLS policies:** Database row-level security policies enforce customer-only access to their own data
  - **Admin RLS:** `get_own_role()` security-definer function returns user's role for RLS comparisons
  
### Session & JWT
- **Type:** Supabase JWT (auth.users table)
- **Storage:** Secure HTTP-only cookies + in-memory Supabase client state
- **Refresh:** Automatic via middleware on every request (if stale)
- **User object claims:** 
  - `sub` (user ID)
  - `email`
  - `email_verified`
  - User metadata from OAuth provider
- **Profile columns for authorization:**
  - **role:** 'customer' | 'admin'
  - **verification_status:** 'not_started' | 'pending_review' | 'verified' | 'rejected' | 'on_hold'
  - **pilot_clearance_status:** 'checkout_required' | 'checkout_requested' | 'checkout_confirmed' | 'checkout_completed_under_review' | 'checkout_payment_required' | 'cleared_to_fly' | 'additional_checkout_required' | 'checkout_reschedule_required' | 'not_currently_eligible'
  - **account_status:** 'active' | 'blocked' | 'archived'
  - **account_lock_reason:** 'checkout_no_show' | null

const TWELVE_HOURS_MS = 12 * 60 * 60 * 1000

function isCheckoutSelfServiceAllowed(startIsoUtc: string, now = new Date()): boolean {
  const startMs = new Date(startIsoUtc).getTime()
  if (Number.isNaN(startMs)) return false
  return now.getTime() < (startMs - TWELVE_HOURS_MS)
}

// ─── Auth guard (no verification requirement) ─────────────────────────────────
// Checkout bookings are created before the user is verified.
// We only require a valid authenticated session and a customer account.

async function requireCustomer() {
  const supabase = await createClient()
  const { data: { user }, error } = await supabase.auth.getUser()
  if (error || !user) throw new Error('Unauthorized')

  const { data: profile } = await supabase
    .from('profiles')
    .select('email, role, pilot_clearance_status, has_night_vfr_rating, has_instrument_rating, account_status, account_lock_reason')
    .eq('id', user.id)
    .single()

  if (!profile) throw new Error('Profile not found')
  if (profile.role !== 'customer') throw new Error('Not a customer account')
  if (isNoShowLockedProfile(profile)) {
    throw new Error('ACCOUNT_BLOCKED: Your account is locked due to a checkout no-show. Please contact OZ Rent A Plane.')
  }

  return { supabase, userId: user.id, profile }
}

async function requireAdmin() {
  const supabase = await createClient()
  const { data: { user }, error } = await supabase.auth.getUser()
  if (error || !user) throw new Error('Unauthorized')

  const { data: profile } = await supabase
    .from('profiles')
    .select('role')
    .eq('id', user.id)
    .single()

  if (!profile || profile.role !== 'admin') throw new Error('Forbidden')
  return { supabase, adminId: user.id }
}

const CUSTOMER_MODIFIABLE_CHECKOUT_STATUSES = ['checkout_requested', 'checkout_confirmed'] as const

function deriveLifecycleFromBookingStatus(status: string) {
  if (status === 'checkout_requested') return 'requested'
  if (status === 'checkout_confirmed') return 'scheduled'
  if (status === 'checkout_completed_under_review' || status === 'checkout_payment_required' || status === 'completed') {
    return 'completed'
  }
  return 'scheduled'
}

function getSydneyDateAndTime(isoUtc: string): { date: string; time: string } {
  const d = new Date(isoUtc)
  const date = d.toLocaleDateString('en-CA', { timeZone: 'Australia/Sydney' })
  const time = d.toLocaleTimeString('en-GB', {
    timeZone: 'Australia/Sydney',
    hour: '2-digit',
    minute: '2-digit',
    hour12: false,
  })
  return { date, time }
}

export async function canModifyCheckout(checkout: {
  booking_type?: string | null
  status?: string | null
  scheduled_start?: string | null
  checkout_lifecycle_status?: string | null
}) {
  if (!checkout || checkout.booking_type !== 'checkout') return false
  if (!checkout.status || !CUSTOMER_MODIFIABLE_CHECKOUT_STATUSES.includes(checkout.status as typeof CUSTOMER_MODIFIABLE_CHECKOUT_STATUSES[number])) {
    return false
  }
  if (!checkout.scheduled_start) return false
  if (!isCheckoutSelfServiceAllowed(checkout.scheduled_start, new Date())) return false
  if (checkout.checkout_lifecycle_status === 'cancelled_by_customer' || checkout.checkout_lifecycle_status === 'cancelled_by_admin' || checkout.checkout_lifecycle_status === 'completed') {
    return false
  }
  return true
}

// ─── Submit checkout request ──────────────────────────────────────────────────

// Server-side validation before calling the RPC:

//      be uploaded and not rejected or expired.
//   2. Time range must be valid and in the future.
//
// The RPC (create_checkout_booking_atomic) then enforces at the database layer:
//   • Clearance status allows a new checkout request
//   • No currently active checkout booking exists
//   • Aircraft is available
//   • No schedule block conflicts

  const requestPath = h.get('x-matched-path') ?? '/dashboard/checkout'
  const requestMethod = h.get('x-http-method-override') ?? 'POST'
  const routeName = 'checkout-submit-debug'
  console.info(`[${routeName}] request received`, {
    host: requestHost,
    path: requestPath,
    method: requestMethod,
    step: 'pre_auth_guard',
  })

  let supabase: Awaited<ReturnType<typeof createClient>>
  let userId = ''
  let profile: Awaited<ReturnType<typeof requireCustomer>>['profile']
  try {
    const authCtx = await requireCustomer()
    supabase = authCtx.supabase
    userId = authCtx.userId
    profile = authCtx.profile
  } catch (error) {
    const message = error instanceof Error ? error.message : 'Unknown auth error'
    const lower = message.toLowerCase()
    const isAuthError = lower.includes('unauthorized') || lower.includes('not authenticated') || lower.includes('invalid jwt')
    const isAccountBlocked = message.startsWith('ACCOUNT_BLOCKED:')
    console.error(`[${routeName}] auth guard failed`, {
      host: requestHost,
      step: 'auth_guard',
      auth_user_exists: false,
      message,
    })
    if (isAccountBlocked) {
      return {
        ok: false,
        type: 'account_blocked',
        message: message.replace(/^ACCOUNT_BLOCKED:\s*/i, ''),
      }
    }
    if (isAuthError) {
      return {
        ok: false,
        type: 'auth',
        message: 'Your session has expired. Please sign in again, then resubmit your checkout request.',
      }
    }
    return { ok: false, type: 'error', message: 'We could not verify your session. Please try again.' }
  }
  const ACCEPTANCE_TEXT = 'I have read and accept the Checkout Terms and Conditions.'
  const profileRecord = profile as Record<string, unknown>
  const safeEmail = typeof profileRecord.email === 'string' ? profileRecord.email : null
  const safeAccountStatus = typeof profileRecord.account_status === 'string' ? profileRecord.account_status : null
  try {
    console.info(`[${routeName}] submit started`, {
      host: requestHost,
      user_id: userId,
      email: safeEmail,
      auth_user_exists: true,

      has_last_flight_date: Boolean(input.last_flight_date),

      host: requestHost,
      user_id: userId,
      email: safeEmail,

      account_status: safeAccountStatus,
      pilot_clearance_status: profile.pilot_clearance_status,
      has_instrument_rating: profile.has_instrument_rating,
    })

  // ── Document gate ──────────────────────────────────────────────────────────
  // Validates all required document fields per document type.
  const { data: docs, error: docsErr } = await supabase
    .from('user_documents')
    .select('document_type, status, expiry_date, issue_date, licence_type, licence_number, medical_class, id_type, document_number')
    .eq('user_id', userId)

  if (docsErr) {
    console.error(`[${routeName}] document query error`, { code: docsErr.code, message: docsErr.message })
    return {
      ok: false,
      type: 'error',
      message: 'Could not load your documents. Please try again.',
    }
  }

  const docsArray = (docs ?? []) as Array<{
    document_type: string
    status: string
    expiry_date: string | null
    issue_date: string | null
    licence_type: string | null
    licence_number: string | null
    medical_class: string | null
    id_type: string | null
    document_number: string | null
  }>

  const requiredDocs = ['pilot_licence', 'medical_certificate', 'photo_id']
  const uploadedDocs = new Map(docsArray.map(d => [d.document_type, d]))

  const missingDocs = requiredDocs.filter(type => !uploadedDocs.has(type))
  if (missingDocs.length > 0) {
    console.warn(`[${routeName}] document validation failed — missing docs`, { missing: missingDocs })
    return {
      ok: false,
      type: 'validation',
      message: `Please upload all required documents before submitting your checkout request: ${missingDocs.join(', ')}`,
    }
  }

  for (const type of requiredDocs) {
    const doc = uploadedDocs.get(type)!
    if (doc.status === 'rejected') {
      console.warn(`[${routeName}] document validation failed — rejected`, { document_type: type })
      return {
        ok: false,
        type: 'validation',
        message: `Your ${type.replace(/_/g, ' ')} was rejected. Please upload a new one.`,
      }
    }
    if (type === 'medical_certificate' && doc.expiry_date) {
      const expiry = new Date(doc.expiry_date)
      if (expiry < new Date()) {
        console.warn(`[${routeName}] document validation failed — expired`, { document_type: type })
        return {
          ok: false,
          type: 'validation',
          message: 'Your medical certificate has expired. Please upload a new one.',
        }
      }
    }
  }

  console.info(`[${routeName}] document validation passed ✓`)

  // ── Schedule validation ────────────────────────────────────────────────────
  if (!input.aircraft_id) {
    return {
      ok: false,
      type: 'validation',
      message: 'Aircraft not found. Please try again.',
    }
  }

  if (!input.scheduled_start_iso_utc) {
    return {
      ok: false,
      type: 'validation',
      message: 'Please select a date and time for your checkout flight.',
    }
  }

  const startMs = new Date(input.scheduled_start_iso_utc).getTime()
  if (Number.isNaN(startMs)) {
    return {
      ok: false,
      type: 'validation',
      message: 'Invalid date/time. Please try again.',
    }
  }

  const now = new Date().getTime()
  if (startMs <= now) {
    return {
      ok: false,
      type: 'validation',
      message: 'Checkout flight time must be in the future.',
    }
  }

  // ── Last flight review date validation ─────────────────────────────────────
  if (!input.last_flight_date) {
    return {
      ok: false,
      type: 'validation',
      message: 'Please enter your last flight review date before submitting your checkout request.',
    }
  }

  try {
    const flightReviewValidation = validateFlightReviewDate(input.last_flight_date)
    if (!flightReviewValidation.valid) {
      console.warn(`[${routeName}] flight review date validation failed`, {
        last_flight_date: input.last_flight_date,
        reason: flightReviewValidation.reason,
      })
      return {
        ok: false,
        type: 'validation',
        message: `Flight review date error: ${flightReviewValidation.reason}`,
      }
    }
  } catch (e: any) {
    console.error(`[${routeName}] flight review validation threw`, { message: e?.message })
    return {
      ok: false,
      type: 'error',
      message: 'Could not validate your flight review date. Please try again.',
    }
  }

  // ── Ratings validation ─────────────────────────────────────────────────────
  if (input.has_night_vfr === undefined || input.has_night_vfr === null) {
    return {
      ok: false,
      type: 'validation',
      message: 'Please confirm your Night VFR rating status before submitting your checkout request.',
    }
  }
  if (input.has_instrument_rating === undefined || input.has_instrument_rating === null) {
    return {
      ok: false,
      type: 'validation',
      message: 'Please confirm your Instrument Rating status before submitting your checkout request.',
    }
  }

  console.info(`[${routeName}] pre-schedule validation passed ✓`)

  // ── Time range validation (Day VFR window) ─────────────────────────────────
  const dayVfrWindow = getDayVfrWindow(input.scheduled_start_iso_utc)
  if (!isWithinDayVfrWindow(input.scheduled_start_iso_utc)) {
    return {
      ok: false,
      type: 'validation',
      message: `Checkout flights must be scheduled between ${dayVfrWindow.start} and ${dayVfrWindow.end} Sydney time.`,
    }
  }

  console.info(`[${routeName}] day VFR validation passed ✓`)

  // ── Availability check ─────────────────────────────────────────────────────
  const availability = await checkAircraftAvailability(
    input.aircraft_id,
    input.scheduled_start_iso_utc,
    input.scheduled_start_iso_utc
  )

  if (!availability.available) {
    console.warn(`[${routeName}] availability check failed`, {
      reason: availability.reason,
      conflicts: availability.conflicts?.length ?? 0,
    })
    return {
      ok: false,
      type: 'availability',
      message: `AVAILABILITY: ${availability.reason || 'This time slot is not available'}`,
    }
  }

  console.info(`[${routeName}] availability check passed ✓`)

  // ── Terms acceptance validation ────────────────────────────────────────────
  if (!input.terms_document_id || !input.terms_version || !input.terms_content_hash) {
    return {
      ok: false,
      type: 'validation',
      message: 'Terms acceptance details were not submitted. Please review and accept the checkout terms.',
    }
  }

  // Verify the terms document is still active
  const { data: termsDoc, error: termsErr } = await supabase
    .from('terms_documents')
    .select('id, version, content_hash, is_active')
    .eq('id', input.terms_document_id)
    .single()

  if (termsErr || !termsDoc) {
    console.error(`[${routeName}] terms document query error`, { code: termsErr?.code, message: termsErr?.message })
    return {
      ok: false,
      type: 'validation',
      message: 'No active checkout terms document found. Please refresh and try again.',
    }
  }

  if (!termsDoc.is_active) {
    console.warn(`[${routeName}] terms document not active`, { terms_id: input.terms_document_id })
    return {
      ok: false,
      type: 'validation',
      message: 'The checkout terms have been updated. Please review and accept the latest version.',
    }
  }

  if (termsDoc.version !== input.terms_version) {
    console.warn(`[${routeName}] terms version mismatch`, {
      submitted_version: input.terms_version,
      current_version: termsDoc.version,
    })
    return {
      ok: false,
      type: 'validation',
      message: 'The checkout terms have been updated. Please review and accept the latest version.',
    }
  }

  if (termsDoc.content_hash && termsDoc.content_hash !== input.terms_content_hash) {
    console.warn(`[${routeName}] terms content hash mismatch`, {
      submitted_hash: input.terms_content_hash,
      current_hash: termsDoc.content_hash,
    })
    return {
      ok: false,
      type: 'validation',
      message: 'The checkout terms have been updated. Please review and accept the latest version.',
    }
  }

  console.info(`[${routeName}] terms validation passed ✓`)

  // ── Create checkout booking via RPC ────────────────────────────────────────
  console.info(`[${routeName}] calling create_checkout_booking_atomic RPC`, {
    aircraft_id: input.aircraft_id,
    scheduled_start: input.scheduled_start_iso_utc,
  })

  const { data: rpcResult, error: rpcErr } = await supabase.rpc('create_checkout_booking_atomic', {
    p_aircraft_id:     input.aircraft_id,
    p_scheduled_start: input.scheduled_start_iso_utc,
    p_customer_notes:  input.customer_notes ?? null,
  })

  if (rpcErr) {
    console.error(`[${routeName}] RPC failed`, { code: rpcErr.code, message: rpcErr.message })
    const lower = rpcErr.message.toLowerCase()
    if (lower.includes('cannot use')) {
      return { ok: false, type: 'validation', message: `Validation: ${rpcErr.message}` }
    }
    if (lower.includes('conflict') || lower.includes('schedule block')) {
      return { ok: false, type: 'availability', message: `AVAILABILITY: ${rpcErr.message}` }
    }
    return { ok: false, type: 'error', message: 'Failed to create checkout booking. Please try again.' }
  }

  if (!rpcResult) {
    console.error(`[${routeName}] RPC returned null result`)
    return { ok: false, type: 'error', message: 'Checkout booking response was empty. Please try again.' }
  }

  const typedResult = rpcResult as Record<string, unknown>
  const bookingId = typedResult.booking_id as string | undefined
  if (!bookingId) {
    console.error(`[${routeName}] RPC result missing booking_id`, { result: rpcResult })
    return { ok: false, type: 'error', message: 'Invalid server response. Please try again.' }
  }

  console.info(`[${routeName}] RPC succeeded ✓`, { booking_id: bookingId })

  // ── Record terms acceptance ────────────────────────────────────────────────
  console.info(`[${routeName}] recording terms acceptance`, { booking_id: bookingId })

  const { error: termsAcceptErr } = await supabase.rpc('record_terms_acceptance', {
    p_user_id:           userId,
    p_booking_id:        bookingId,
    p_terms_document_id: input.terms_document_id,
    p_terms_version:     input.terms_version,
    p_content_hash:      input.terms_content_hash,
    p_acceptance_context: 'checkout_submit',
  })

  if (termsAcceptErr) {
    console.warn(`[${routeName}] record_terms_acceptance RPC failed (non-fatal)`, {
      code: termsAcceptErr.code,
      message: termsAcceptErr.message,
    })
    return {
      ok: false,
      type: 'error',
      message: 'Your checkout request was created, but we could not record your terms acceptance. Please contact support.',
    }
  }

  console.info(`[${routeName}] terms acceptance recorded ✓`)

  // ── Update profile with ratings & flight date ──────────────────────────────
  console.info(`[${routeName}] updating profile with ratings and flight date`)

  const { error: profileUpdateErr } = await supabase
    .from('profiles')
    .update({
      has_night_vfr_rating: input.has_night_vfr ?? null,
      has_instrument_rating: input.has_instrument_rating ?? null,
      last_flight_date: input.last_flight_date ?? null,
    })
    .eq('id', userId)

  if (profileUpdateErr) {
    console.warn(`[${routeName}] profile update failed (non-critical)`, {
      code: profileUpdateErr.code,
      message: profileUpdateErr.message,
    })
  } else {
    console.info(`[${routeName}] profile updated ✓`)
  }

  // ── Send notification ──────────────────────────────────────────────────────
  try {
    const { data: booking } = await supabase
      .from('bookings')
      .select('scheduled_start, scheduled_end, created_at')
      .eq('id', bookingId)
      .single()

    if (booking) {
      await notifyCheckoutRequestSubmitted(supabase, userId, bookingId, booking.scheduled_start as string)
    }
  } catch (notifErr: any) {
    console.warn(`[${routeName}] notification failed (non-critical)`, { message: notifErr?.message })
  }

  // ── Clear cache & return success ───────────────────────────────────────────
  revalidatePath('/dashboard/checkout')
  revalidatePath('/dashboard')

  console.info(`[${routeName}] checkout submission completed successfully ✓`, { booking_id: bookingId })

  return {
    ok: true,
    type: 'success',
    booking_id: bookingId,
  }
  } catch (error) {
    const message = error instanceof Error ? error.message : 'Unknown error'
    console.error(`[${routeName}] fatal error`, {
      message,
      stack: error instanceof Error ? error.stack : undefined,
    })
    return {
      ok: false,
      type: 'error',
      message: 'An unexpected error occurred. Please try again.',
    }
  }
}

'use server'

import { createClient } from '@/lib/supabase/server'
import type { DocumentType } from '@/lib/supabase/types'

const MAX_FILE_SIZE = 10 * 1024 * 1024 // 10MB

export async function uploadVerificationDocument(
  documentType: DocumentType,
  fileName: string,
  fileData: Buffer
): Promise<{ success: boolean; error?: string }> {
  const supabase = await createClient()
  const { data: { user }, error: authError } = await supabase.auth.getUser()

  if (authError || !user) {
    return { success: false, error: 'Unauthorized' }
  }

  if (fileData.length > MAX_FILE_SIZE) {
    return { success: false, error: `File size exceeds 10MB limit` }
  }

  try {
    const timestamp = Date.now()
    const storagePath = `${user.id}/${documentType}-${timestamp}.pdf`

    const { error: uploadError } = await supabase.storage
      .from('verification_documents')
      .upload(storagePath, fileData, {
        contentType: 'application/pdf',
        upsert: false,
      })

    if (uploadError) {
      console.error('[uploadVerificationDocument] upload error', uploadError)
      return { success: false, error: 'Failed to upload file. Please try again.' }
    }

    // Record in database
    const { error: dbError } = await supabase
      .from('user_documents')
      .upsert({
        user_id: user.id,
        document_type: documentType,
        file_name: fileName,
        storage_path: storagePath,
        status: 'uploaded',
        uploaded_at: new Date().toISOString(),
      }, {
        onConflict: 'user_id,document_type',
      })

    if (dbError) {
      console.error('[uploadVerificationDocument] db error', dbError)
      return { success: false, error: 'Failed to record document. Please try again.' }
    }

    return { success: true }
  } catch (error) {
    console.error('[uploadVerificationDocument] error', error)
    return { success: false, error: 'An error occurred during upload.' }
  }
}

    const endDate = new Date(input.scheduledEnd)
    const bufferMinutes = 30

      .from('schedule_blocks')

          created_by_user_id: user.id,

          created_by_user_id: user.id,
          created_by_role: 'customer',
          is_public_visible: false,
          status: 'active',
        },
      ])

    if (blockErr) {
      console.warn('[createBookingRequest] schedule block creation failed (non-fatal)', blockErr)
    }

    revalidatePath('/dashboard/bookings')
    return { success: true, bookingId: booking.id }
  } catch (error) {
    console.error('[createBookingRequest] error', error)
    return { success: false, error: 'An error occurred' }
  }
}

export async function submitFlightRecord(input: {
  bookingId: string
  tachoDelta: number
  customerNotes?: string
  signature?: string
}): Promise<{ success: boolean; error?: string }> {
  const supabase = await createClient()
  const { data: { user }, error: authErr } = await supabase.auth.getUser()
  if (authErr || !user) return { success: false, error: 'Unauthorized' }

  try {
    const { data: booking } = await supabase
      .from('bookings')
      .select('id, aircraft_id')
      .eq('id', input.bookingId)
      .single()

    if (!booking) return { success: false, error: 'Booking not found' }

    const { error: recordErr } = await supabase
      .from('flight_records')
      .insert({
        booking_id: input.bookingId,
        aircraft_id: booking.aircraft_id,
        date: new Date().toISOString().split('T')[0],
        tacho_total: input.tachoDelta,
        customer_notes: input.customerNotes,
        signature_type: input.signature ? 'typed' : 'none',
        signature_value: input.signature,
        submitted_by_user_id: user.id,
        submitted_at: new Date().toISOString(),
        status: 'submitted',
      })

    if (recordErr) {
      console.error('[submitFlightRecord] insert error', recordErr)
      return { success: false, error: 'Failed to submit flight record' }
    }

    // Update booking status
    await supabase
      .from('bookings')
      .update({
        status: 'pending_post_flight_review',
      })
      .eq('id', input.bookingId)

    revalidatePath(`/dashboard/bookings/${input.bookingId}`)
    return { success: true }
  } catch (error) {
    console.error('[submitFlightRecord] error', error)
    return { success: false, error: 'An error occurred' }
  }
}
